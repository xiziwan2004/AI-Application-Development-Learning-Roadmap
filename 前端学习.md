# 前端学习

## 1. JavaScript 语法基础

### 1.1 变量：`let` / `const` / `var`

- `const`：声明常量，**不能重新赋值**（对象内部属性仍可修改），优先使用。
- `let`：块级作用域，需要重新赋值时使用。
- `var`：函数作用域，存在变量提升，**现代代码中避免使用**。

```js
const MAX_COUNT = 10;
let current = 0;

current = current + 1;      // ✅ let 可以重新赋值
// MAX_COUNT = 20;          // ❌ TypeError: Assignment to constant variable

const user = { name: 'Ada', age: 30 };
user.age = 31;              // ✅ 对象属性可改（引用未变）
// user = {};               // ❌ 不能重新指向新对象

// 块级作用域
{
  let inner = '只在这个块里可见';
  console.log(inner);
}
// console.log(inner);      // ❌ ReferenceError

// 基本类型 vs 引用类型
let a = 1;
let b = a;                  // 复制值
b = 2;
console.log(a, b);          // 1 2

let o1 = { n: 1 };
let o2 = o1;                // 复制引用
o2.n = 9;
console.log(o1.n);          // 9
```

### 1.2 模板字符串与解构

```js
const user = { name: 'Ada', age: 30, city: 'London' };

// 模板字符串
console.log(`你好，${user.name}，你今年 ${user.age} 岁`);

// 对象解构 + 默认值 + 重命名
const { name, age, city = '未知', name: nickname } = user;
console.log(name, age, city, nickname);

// 剩余属性
const { city: _c, ...rest } = user;
console.log(rest);          // { name: 'Ada', age: 30 }

// 数组解构
const [first, second = 100, ...others] = [1, 2, 3, 4];
console.log(first, second, others);   // 1 2 [3, 4]

// 交换变量
let x = 1, y = 2;
[x, y] = [y, x];
```

### 1.3 函数

函数声明、函数表达式、箭头函数；注意箭头函数的 `this` 与 `arguments` 差异。

```js
// 函数声明（有提升）
function add(a, b = 0) {
  return a + b;
}

// 函数表达式（不提升）
const sub = function (a, b) {
  return a - b;
};

// 箭头函数：只有一行表达式时可省略 return
const mul = (a, b) => a * b;
const getUser = () => ({ name: 'Ada' });   // 返回对象需包一层小括号

// 剩余参数
function sum(...nums) {
  return nums.reduce((total, n) => total + n, 0);
}
console.log(sum(1, 2, 3));   // 6

// 箭头函数没有自己的 this，会继承外层的 this
const counter = {
  count: 0,
  start() {
    setInterval(() => {
      this.count++;          // ✅ this 指向 counter
      console.log(this.count);
    }, 1000);
  },
};
counter.start();

// 闭包：函数记住并访问其定义时的作用域
function createCounter() {
  let value = 0;
  return {
    inc: () => ++value,
    get: () => value,
  };
}
const c = createCounter();
c.inc();
c.inc();
console.log(c.get());        // 2
```

### 1.4 对象

```js
const name = 'Ada';

const user = {
  name,                       // 简写属性
  age: 30,
  role: 'engineer',
  greet() {                   // 简写方法
    return `Hi, I'm ${this.name}`;
  },
  get info() {                // getter
    return `${this.name} (${this.role})`;
  },
};

console.log(user.greet());    // Hi, I'm Ada
console.log(user.info);       // Ada (engineer)

// 动态属性名
const key = 'level';
user[key] = 3;

// 展开与合并（浅拷贝）
const copy = { ...user, age: 31 };
const merged = { ...user, ...{ country: 'UK' } };

// 遍历
console.log(Object.keys(user));     // ['name', 'age', 'role', ...]
console.log(Object.values(user));
console.log(Object.entries(user));  // [['name', 'Ada'], ...]

// 可选链与空值合并：安全访问深层属性
const apiResult = { data: { user: null } };
console.log(apiResult?.data?.user?.name);           // undefined（不报错）
console.log(apiResult?.data?.user?.name ?? '匿名');  // 匿名
```

### 1.5 数组常用操作与 `map` / `filter` / `find`

| 方法 | 作用 | 返回值 |
| --- | --- | --- |
| `map` | 逐个转换元素 | 新数组（长度不变） |
| `filter` | 筛选元素 | 新数组（可能为空） |
| `find` | 找第一个匹配的元素 | 元素本身或 `undefined` |
| `findIndex` | 找第一个匹配的索引 | 索引或 `-1` |
| `reduce` | 聚合为单个值 | 任意类型 |
| `some` / `every` | 是否存在 / 是否全部满足 | `boolean` |

```js
const users = [
  { id: 1, name: 'Ada', age: 30, active: true },
  { id: 2, name: 'Bob', age: 17, active: false },
  { id: 3, name: 'Cara', age: 25, active: true },
];

// map：取出所有名字
const names = users.map(u => u.name);
console.log(names);                        // ['Ada', 'Bob', 'Cara']

// map：生成新对象
const cards = users.map(u => ({
  id: u.id,
  label: `${u.name}（${u.age}岁）`,
}));

// filter：成年的活跃用户
const adults = users.filter(u => u.age >= 18 && u.active);
console.log(adults.map(u => u.name));      // ['Ada', 'Cara']

// find：查找单个元素（找不到返回 undefined）
const bob = users.find(u => u.name === 'Bob');
console.log(bob?.id);                      // 2

// 链式调用：先筛选再转换
const activeNames = users
  .filter(u => u.active)
  .map(u => u.name.toUpperCase());
console.log(activeNames);                  // ['ADA', 'CARA']

// reduce：求和 / 分组
const totalAge = users.reduce((sum, u) => sum + u.age, 0);
const groupByActive = users.reduce((acc, u) => {
  const k = u.active ? 'active' : 'inactive';
  (acc[k] ||= []).push(u.name);
  return acc;
}, {});
console.log(totalAge, groupByActive);
// 72 { active: ['Ada', 'Cara'], inactive: ['Bob'] }

// some / every
console.log(users.some(u => u.age < 18));   // true
console.log(users.every(u => u.age > 18));  // false
```

> 提醒：`map` / `filter` / `find` 都**不会修改原数组**；`sort` / `reverse` / `splice` 会修改原数组，需要副本时先用 `[...arr]`。

```js
// 其他高频方法
const nums = [3, 1, 2];

console.log([...nums].sort((a, b) => a - b));  // [1, 2, 3]（不污染原数组）
console.log(nums.includes(2));                 // true
console.log(nums.flatMap(n => [n, n * 2]));    // [3, 6, 1, 2, 2, 4]
console.log(Array.from({ length: 3 }, (_, i) => i)); // [0, 1, 2]
```

### 1.6 Promise

Promise 表示“未来才会有结果”的异步操作，有三种状态：`pending` → `fulfilled` / `rejected`，一旦落定不可再变。

```js
// 创建 Promise
function delay(ms, value) {
  return new Promise((resolve, reject) => {
    if (ms < 0) return reject(new Error('ms 不能为负数'));
    setTimeout(() => resolve(value), ms);
  });
}

// then / catch / finally
delay(300, '完成')
  .then(result => {
    console.log(result);        // 完成
    return result + '!';        // 返回普通值会被包装成 Promise
  })
  .then(v => console.log(v))    // 完成!
  .catch(err => console.error('出错了：', err.message))
  .finally(() => console.log('无论成功失败都会执行'));

// 并发控制
const p1 = delay(100, 'A');
const p2 = delay(200, 'B');

Promise.all([p1, p2]).then(v => console.log(v));        // ['A', 'B']（任一失败即整体失败）
Promise.allSettled([p1, p2]).then(v => console.log(v)); // 每项含 status/result
Promise.race([p1, p2]).then(v => console.log(v));       // 'A'（最快完成者）
Promise.any([p1, p2]).then(v => console.log(v));        // 'A'（第一个成功者）

// 常见写法：包装回调式 API
function readFileAsync(path) {
  return new Promise((resolve, reject) => {
    fs.readFile(path, 'utf8', (err, data) => {
      if (err) reject(err);
      else resolve(data);
    });
  });
}
```

### 1.7 `async` / `await`

`async` 函数总是返回 Promise；`await` 等待 Promise 落定并取出结果，让异步代码读起来像同步代码。

```js
async function getUser(id) {
  // await 等待异步结果
  const user = await delay(100, { id, name: 'Ada' });
  return user;                 // 自动包装为 Promise
}

// 调用
getUser(1).then(u => console.log(u));

// 在 async 函数内部使用 try/catch 捕获错误
async function loadUserSafe(id) {
  try {
    const user = await getUser(id);
    console.log('拿到用户：', user.name);
    return user;
  } catch (err) {
    console.error('加载失败：', err.message);
    return null;
  } finally {
    console.log('请求结束');
  }
}

// 顺序执行 vs 并行执行
async function sequential() {
  const a = await delay(100, 'A');   // 先等 100ms
  const b = await delay(100, 'B');   // 再等 100ms，共约 200ms
  return [a, b];
}

async function parallel() {
  const [a, b] = await Promise.all([  // 同时开始，共约 100ms
    delay(100, 'A'),
    delay(100, 'B'),
  ]);
  return [a, b];
}

// 注意：forEach 里的 await 不会按预期等待，用 for...of 或 Promise.all
async function bad() {
  [1, 2, 3].forEach(async n => {
    await delay(10, n);                // ❌ 外层不会等待
  });
}

async function good() {
  for (const n of [1, 2, 3]) {
    await delay(10, n);                // ✅ 顺序等待
  }
  await Promise.all([1, 2, 3].map(n => delay(10, n))); // ✅ 并行等待
}
```

### 1.8 `fetch`

`fetch` 是浏览器内置的 HTTP 请求 API，返回 Promise，**只有网络错误才会 reject**；`404/500` 需要通过 `response.ok` 判断。

```js
// GET 请求
async function getUsers() {
  const res = await fetch('https://jsonplaceholder.typicode.com/users');

  if (!res.ok) {
    throw new Error(`请求失败：${res.status} ${res.statusText}`);
  }

  const data = await res.json();   // 也可以用 .text() / .blob()
  return data;
}

getUsers().then(users => console.log(users.length));

// POST 请求（JSON）
async function createUser(payload) {
  const res = await fetch('https://jsonplaceholder.typicode.com/users', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      // 'Authorization': `Bearer ${token}`,
    },
    body: JSON.stringify(payload),
  });

  if (!res.ok) throw new Error(`创建失败：${res.status}`);
  return res.json();
}

// 超时控制（AbortController）
async function fetchWithTimeout(url, ms = 5000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), ms);
  try {
    const res = await fetch(url, { signal: controller.signal });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    return await res.json();
  } finally {
    clearTimeout(timer);
  }
}

// 带重试的简单封装
async function fetchWithRetry(url, retries = 3) {
  for (let i = 0; i < retries; i++) {
    try {
      return await fetchWithTimeout(url);
    } catch (err) {
      if (i === retries - 1) throw err;
      await delay(500 * (i + 1));
    }
  }
}
```

> 流式响应的读取方式（AI 应用里很常用）：

```js
const res = await fetch('/api/chat', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ message: '你好' }),
});

const reader = res.body.getReader();
const decoder = new TextDecoder('utf-8');

while (true) {
  const { done, value } = await reader.read();
  if (done) break;
  process.stdout.write(decoder.decode(value, { stream: true }));
}
```

### 1.9 `import` / `export`（ES Module）

```js
// ---------- math.js ----------
export const PI = 3.14159;                    // 命名导出

export function add(a, b) {
  return a + b;
}

export default function multiply(a, b) {      // 默认导出（一个模块只能有一个）
  return a * b;
}

// ---------- utils.js ----------
// 先定义再统一导出
const formatDate = d => d.toISOString().slice(0, 10);
const VERSION = '1.0.0';

export { formatDate, VERSION };

// 导出时的重命名
export { VERSION as APP_VERSION };
```

```js
// ---------- main.js ----------
import multiply from './math.js';                 // 默认导入（名字随意）
import { add, PI } from './math.js';              // 命名导入
import { add as plus } from './math.js';          // 重命名
import * as math from './math.js';                // 整体导入
import { formatDate } from './utils.js';

console.log(multiply(2, 3));       // 6
console.log(add(1, 2), PI);        // 3 3.14159
console.log(plus(1, 2));           // 3
console.log(math.add(1, 2));       // 3
console.log(formatDate(new Date()));

// 动态导入：按需加载，返回 Promise（做代码分割）
async function loadChart() {
  const { Chart } = await import('./chart.js');
  return new Chart();
}
```

在浏览器中生效需要给 `<script>` 加上 `type="module"`：

```html
<script type="module" src="./main.js"></script>
```

与 Node.js 的 CommonJS 对比：

```js
// CommonJS（Node 传统写法）
const fs = require('fs');
const { add } = require('./math.js');
module.exports = { add, PI: 3.14159 };
```

| 特性 | ES Module | CommonJS |
| --- | --- | --- |
| 语法 | `import` / `export` | `require` / `module.exports` |
| 加载 | 静态解析，编译期确定 | 运行时加载 |
| 场景 | 浏览器、Vite/Webpack、`"type": "module"` | 老版 Node 项目 |
